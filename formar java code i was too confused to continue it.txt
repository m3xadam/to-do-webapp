function getBucket(dueDate){
    const today = new Date().toISOString().split('T')[0];
    if (dueDate < today) return "Overdue";
    if (dueDate === today) return "Today";

    return "Upcoming";

}

function updateCounters(){
    const total = tasks.length;
    const done = tasks.filter(t=> t.done).length;
    const overdue = tasks.filter(t=> !t.done && getBucket(t.dueDate)=== "Overdue").length;
    const pending = total - done;

    document.getElementById("counter1").textContent = total;
    document.getElementById("counter2").textContent = done;
    document.getElementById("counter3").textContent = overdue;
    document.getElementById("counter4").textContent = pending;
}
function render(){
    list.innerHTML = '';

    taskGroup.forEach(function(group){

        let heading = document.createElement("h2");
        heading.textContent = group.title;

        let deleteGroupBtn = document.createElement("button");
        deleteGroupBtn.textContent = "Delete Group";

        deleteGroupBtn.addEventListener('click', function(){
            let index = taskGroup.indexOf(group);
            taskGroup.splice(index,1);

            render();
        })

        heading.appendChild(deleteGroupBtn);

        let ul = document.createElement("ul");

        group.tasks.forEach(function(task){

            let li = document.createElement("li");

            let deletebtn = document.createElement("button");
            deletebtn.textContent = "Delete";
            
            deletebtn.addEventListener("click", function() {
                let index = group.tasks.indexOf(task);
                 group.tasks.splice(index, 1);

                render();
            });


            let toggle = document.createElement("input");
            toggle.type = "checkbox";
            toggle.checked = task.done;

            let event = document.createElement("span");
            event.textContent = task.text;

            if (task.done) {
                event.style.textDecoration = "line-through";
            }


            toggle.addEventListener("change", function() {

                task.done = toggle.checked;

                if (task.done) {
                    event.style.textDecoration = "line-through";
                } else {
                    event.style.textDecoration = "none";
                }
            });

            li.appendChild(toggle);
            li.appendChild(event);
            li.appendChild(deletebtn);
            
            ul.appendChild(li);
        });

        list.appendChild(heading);
        list.appendChild(ul);
    });

submit.addEventListener("click", function(){

    
    if( userInput.value === ''){
    
        alert("field cannot be empty")
    
        return;
    }
    
    if(dateInput.value === ''){
        alert('please pick a date');
        return;
    }

    taskGroup[0].title

    taskGroup[0].tasks.push({
        id:Date.now(),
        text: userInput.value,
        dueDate:dateInput.value,

    });

    userTaskInfo.style.display = "none";

    userInput.value = "";
    dateInput.value = '';
    render();
})
